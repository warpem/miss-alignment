1"""Pretrained miss-alignment models from easymode. 

2 

3The models were trained by Mart So-Last on 54 datasets as part of easymode 

4(https://mgflast.github.io/easymode/, https://github.com/mgflast/easymode) and are 

5distributed via Hugging Face under GPL-3.0. They are downloaded on first use and 

6cached, never bundled with miss-alignment. 

7 

8The download is pinned to a Hugging Face commit and verified against a SHA-256 

9checksum: the checkpoints are unpickled by Lightning, so only a known archive is 

10ever unpacked. Updating the models means bumping these constants, which also 

11changes the cache directory so earlier results stay reproducible. 

12""" 

13 

14import hashlib 

15import os 

16import shutil 

17import tempfile 

18import urllib.request 

19import zipfile 

20from pathlib import Path 

21from typing import Optional 

22 

23from tqdm import tqdm 

24 

25from ._cli import cli 

26 

27PRETRAINED_REVISION = "d4e9a07ef0ea2695c05ebd7b0e2c49cb50d3da26" 

28PRETRAINED_URL = ( 

29 "https://huggingface.co/mgflast/easymode-v2/resolve/" 

30 f"{PRETRAINED_REVISION}/missalignment/easymode_missalignment.zip" 

31) 

32PRETRAINED_SHA256 = "7cae259abf7c8c11c90e7a5ae3a588101c5ce1b21d6abd8710e4761e1527b43b" 

33ARCHIVE_ROOT = "easymode_missalignment" 

34 

35# Schedule the pretrained models were trained for: model iterN expects step N. 

36PRETRAINED_ITERATION_SETTINGS = [ 

37 {"downsample": 3, "alignment": "anchoring"}, 

38 {"downsample": 2, "alignment": "anchoring"}, 

39 {"downsample": 1, "alignment": "global"}, 

40 {"downsample": 1, "alignment": "global"}, 

41 {"downsample": 1, "alignment": [3, 3]}, 

42 {"downsample": 1, "alignment": [3, 3]}, 

43 {"downsample": 1, "alignment": [3, 3]}, 

44 {"downsample": 1, "alignment": [3, 3]}, 

45] 

46N_PRETRAINED_ITERATIONS = len(PRETRAINED_ITERATION_SETTINGS) 

47 

48ATTRIBUTION = ( 

49 "Models were pretrained on a subset of the " 

50 "easymode dataset by Mart So-Last, see: " 

51 "https://doi.org/10.64898/2026.05.19.726344" 

52)

54# Overrides the cache location, e.g. to a shared filesystem for cluster workers. 

55CACHE_DIR_ENV_VAR = "MISS_ALIGNMENT_CACHE_DIR"

58def default_cache_dir() -> Path: 

59 """Cache root: ``$MISS_ALIGNMENT_CACHE_DIR``, else the XDG cache directory.""" 

60 if os.environ.get(CACHE_DIR_ENV_VAR): 

61 return Path(os.environ[CACHE_DIR_ENV_VAR]) 

62 xdg_cache = os.environ.get("XDG_CACHE_HOME") or Path.home() / ".cache" 

63 return Path(xdg_cache) / "miss-alignment" 

64 

65 

66def pretrained_model_dir(cache_dir: Optional[Path] = None) -> Path: 

67 """Directory holding ``iter1..iterN/model.ckpt`` for the pinned revision.""" 

68 cache_dir = default_cache_dir() if cache_dir is None else Path(cache_dir) 

69 return cache_dir / "models" / f"easymode-{PRETRAINED_REVISION[:7]}" 

70 

71 

72def _is_complete(model_dir: Path) -> bool: 

73 return all( 

74 (model_dir / f"iter{i}" / "model.ckpt").is_file() 

75 for i in range(1, N_PRETRAINED_ITERATIONS + 1) 

76 ) 

77 

78 

79def _download(url: str, destination: Path) -> None: 

80 with urllib.request.urlopen(url, timeout=60) as response: 

81 total = int(response.headers.get("Content-Length") or 0) or None 

82 with ( 

83 open(destination, "wb") as f, 

84 tqdm(total=total, unit="B", unit_scale=True, desc="download") as bar, 

85 ): 

86 while chunk := response.read(1 << 20): 

87 f.write(chunk) 

88 bar.update(len(chunk)) 

89 

90 

91def _sha256(path: Path) -> str: 

92 digest = hashlib.sha256() 

93 with open(path, "rb") as f: 

94 while chunk := f.read(1 << 20): 

95 digest.update(chunk) 

96 return digest.hexdigest() 

97 

98 

99def get_pretrained_models( 

100 cache_dir: Optional[Path] = None, 

101 url: str = PRETRAINED_URL, 

102 sha256: str = PRETRAINED_SHA256, 

103) -> Path: 

104 """Return the cached pretrained model directory, downloading it if needed. 

105 

106 Once cached, no network access happens. The archive is downloaded and 

107 unpacked into a temporary directory next to the cache and only moved into 

108 place when complete, so an interrupted download never leaves a partial 

109 cache and concurrent runs cannot corrupt each other. 

110 """ 

111 model_dir = pretrained_model_dir(cache_dir) 

112 if _is_complete(model_dir): 

113 return model_dir 

114 

115 print(f"Downloading pretrained models to {model_dir}") 

116 model_dir.parent.mkdir(parents=True, exist_ok=True) 

117 with tempfile.TemporaryDirectory(dir=model_dir.parent, prefix=".tmp-") as tmp: 

118 tmp = Path(tmp) 

119 archive = tmp / "models.zip" 

120 _download(url, archive) 

121 

122 actual_sha256 = _sha256(archive) 

123 if actual_sha256 != sha256: 

124 raise RuntimeError( 

125 f"Checksum mismatch for pretrained models downloaded from {url}: " 

126 f"expected {sha256}, got {actual_sha256}." 

127 ) 

128 

129 with zipfile.ZipFile(archive) as zf: 

130 zf.extractall(tmp / "extracted") 

131 extracted = tmp / "extracted" / ARCHIVE_ROOT 

132 if not _is_complete(extracted): 

133 raise RuntimeError( 

134 f"Pretrained model archive from {url} does not contain " 

135 f"iter1..iter{N_PRETRAINED_ITERATIONS}/model.ckpt." 

136 ) 

137 

138 # a stale incomplete directory (e.g. manually damaged) is replaced 

139 if model_dir.exists() and not _is_complete(model_dir): 

140 shutil.rmtree(model_dir) 

141 try: 

142 extracted.rename(model_dir) 

143 except OSError: 

144 # another process finished the same download first 

145 if not _is_complete(model_dir): 

146 raise 

147 

148 return model_dir 

149

151@cli.command(name="download-models") 

152def download_models() -> None: 

153 """Download the pretrained easymode models used by `infer` by default. 

154 

155 Useful on clusters where compute nodes have no internet access: run this 

156 once on a login node. The cache must be on a filesystem visible to the 

157 nodes that run `infer`. The cache defaults to ~/.cache/miss-alignment; set 

158 MISS_ALIGNMENT_CACHE_DIR to place it elsewhere (for both commands). 

159 """ 

160 model_dir = get_pretrained_models() 

161 print(ATTRIBUTION) 

162 print(f"Pretrained models available at: {model_dir}")
